cmake_minimum_required(VERSION 3.16)
project(xovium_unreal_terrain VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(CURL REQUIRED)
find_package(JPEG REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgcodecs imgproc)
file(GLOB TERRAIN CONFIGURE_DEPENDS "vendor/terrain/src/core/*.cc")
add_library(xovium_terrain SHARED src/terrain.cpp src/terrain_image.cpp ${TERRAIN})
target_include_directories(xovium_terrain PRIVATE include vendor/terrain/include ${OpenCV_INCLUDE_DIRS} ${JPEG_INCLUDE_DIRS})
target_link_libraries(xovium_terrain PRIVATE Threads::Threads ${CMAKE_DL_LIBS} CURL::libcurl ${OpenCV_LIBS})
target_compile_options(xovium_terrain PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(xovium_terrain PRIVATE XOVIUM_PRIVATE_JPEG=1 XOVIUM_JPEG_LIBRARY="${JPEG_LIBRARY_RELEASE}")
set_target_properties(xovium_terrain PROPERTIES OUTPUT_NAME xovium_dynamic_terrain INSTALL_RPATH_USE_LINK_PATH TRUE)
target_link_options(xovium_terrain PRIVATE "LINKER:-Bsymbolic-functions")
install(TARGETS xovium_terrain LIBRARY DESTINATION Binaries/Linux)
set(TERRAIN_TESTS_AVAILABLE OFF)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../tests/native_abi_test.cpp")
  set(TERRAIN_TESTS_AVAILABLE ON)
endif()
option(BUILD_TESTING "Build native terrain tests (repository checkouts only)" ${TERRAIN_TESTS_AVAILABLE})
if(BUILD_TESTING AND NOT TERRAIN_TESTS_AVAILABLE)
  message(FATAL_ERROR "Tests are not in source releases; use BUILD_TESTING=OFF or a repository checkout")
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(terrain_abi_test ../tests/native_abi_test.cpp)
  target_include_directories(terrain_abi_test PRIVATE include)
  target_link_libraries(terrain_abi_test PRIVATE xovium_terrain Threads::Threads)
  target_compile_options(terrain_abi_test PRIVATE -UNDEBUG)
  add_test(NAME terrain_abi COMMAND terrain_abi_test)
  add_executable(terrain_image_test ../tests/native_image_test.cpp)
  target_include_directories(terrain_image_test PRIVATE include ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(terrain_image_test PRIVATE xovium_terrain ${OpenCV_LIBS})
  target_compile_options(terrain_image_test PRIVATE -UNDEBUG)
  add_test(NAME terrain_image COMMAND terrain_image_test)
  add_executable(terrain_ownership_test ../tests/native_ownership_test.cpp)
  target_include_directories(terrain_ownership_test PRIVATE include vendor/terrain/include ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(terrain_ownership_test PRIVATE xovium_terrain Threads::Threads)
  target_compile_options(terrain_ownership_test PRIVATE -UNDEBUG)
  add_test(NAME terrain_ownership COMMAND terrain_ownership_test)
  set_tests_properties(terrain_abi terrain_image terrain_ownership PROPERTIES TIMEOUT 30)
endif()
